```python
class Solution:
    def diameterOfBinaryTree(self, root: Optional[TreeNode]) -> int:
        ans = 0
        def dfs(node):
            if node is None:
                return -1
            l_depth = dfs(node.left)
            r_depth = dfs(node.right)
            nonlocal ans
            ans = max(ans, l_depth + r_depth + 2)
```

```python
class Solution(object):
    def findKthLargest(self, nums, k):
        n = len(nums)
        self.build_maxHeap(nums)
        for i in range(k-1):
            nums[0], nums[n-1-i] = nums[n-1-i], nums[0]
            self.down(nums, 0, n-1-i-1)
        return nums[0]
    
    def build_maxHeap(self, nums):
        n =len(nums)
        for root in range(n//2 , -1 , -1):
            self.down(nums, root, n-1)

    def down(self, nums, root, hi):
        # 存一手
        t = nums[root]
        # 左孩子
        child = 2 * root + 1 
        while child <= hi:
            # child拿大的那个
            if child +1 <= hi and nums[child]< nums[child+1]:
                child +=1
            # 本来root大就不换了
            if nums[child] < t:
                break
            # 换 
            nums[root] = nums[child]
            root = child
            child = root *2 +1
        nums[root] = t
```

```python
class Solution(object):
    def canFinish(self, numCourses, prerequisites):
       # 首先第一步构建graph，python里面是要dict字典来构造
       graph = defaultdict(list)
       indegree = [0] * numCourses # 入度一开始大家都是0

       # 然后正式把prerequisite里面的关系写到我们的graph里面
       for dest, src in prerequisites: # 这里dest是destination，src是source缩写，方便理解是src->dest
           graph[src].append(dest) # [src:[dest1,dest2,dest3,...]]
           indegree[dest] += 1 # 更新需要src作为先修课的课程，给他们入度加1
       
       # 所以我们先构造一个queue来存储这些indegree==0
       queue = deque([i for i in range(numCourses) if indegree[i] == 0])

       # 再下一步我们开始学习这些indegree为0的课程：
       learned_course = 0
       while queue:
           course = queue.popleft() # 我们要用队列FIFO，从而实现BFS，所以popleft
           learned_course += 1
           
           # 对于是当前要学习的course是自己先修课的那些next_course，我们找出来
           for next_course in graph[course]:
               indegree[next_course] -= 1 # 给他们入度-1，意味着他们先修课要求-1
               if indegree[next_course] == 0: # 如果这些next course已经没有先修课了
                   queue.append(next_course) # 说明下一次循环他们就可以被学习了，加入队列

       return learned_course == numCourses
```

```python
class Solution(object):
    def numIslands(self, grid):
        count = 0
        if not grid: return 0

        # 递归找陆地，越界或者不是陆地就直接返回
        def dfs(i,j):
            if i<0 or i>=len(grid) or j<0 or j>=len(grid[0]) or grid[i][j] != "1":
                return
            # 标记已遍历过
            grid[i][j] = "2"
            dfs(i-1,j)
            dfs(i+1,j)
            dfs(i,j-1)
            dfs(i,j+1)

        # 找到路径就四个方向找，直到没有陆地，最后加1
        for i in range(len(grid)):
            for j in range(len(grid[0])):
                if grid[i][j] == "1":
                    dfs(i,j)
                    count+=1
        return count
```

```python
class Solution(object):

            return 0
        
        # 子问题是如何偷K间房子 也就是n[k-1]
```

### LRU实现

```python

class Node:
    __slots__ = 'prev', 'next', 'key', 'value'

    def __init__(self, key=0, value=0):
        self.key = key
        self.value = value

class LRUCache:
    def __init__(self, capacity):
        self.capacity = capacity
        self.dummy = Node()
        self.dummy.pre = self.dummy
        self.dummy.next = self.dummy
        self.key2node = {}

    # 获取节点，把节点移到链表头部
    def get_node(self, key):
        if key not in self.key2node:
            return None
        node = self.key2node[key]
        # 先抽
        self.remove(node)
        # 再推到头部
        self.push_front(node)
        return node
    
    def get(self, key):
        node = self.get_node(key)
        return node.value if node else -1

    def put(self, key, value):
        # 如果有就先移
        node = self.get_node(key)
        if node:
            node.value = value
            return
        # 没有就新建
        self.key2node[key] = node = Node(key, value)
        self.push_front(node)
        # 超额 就删除最后一个
        if len(self.key2node) > self.capacity:
            back = self.dummy.prev
            del self.key2node[back.key]
            self.remove(back)

    # 删除
    def remove(self, x):
        # 先断前
        x.prev.next = x.next
        # 断后
        x.next.prev = x.prev

    # 推到最前
    def push_front(self, x):
        # 先管自己
        x.prev = self.dummy
        x.next = self.dummy.next
        # 再管插入后的前后
        x.prev.next = x
        x.next.prev = x

```

```python
            for j in range(i-1, max(i-max_len-1, -1), -1):
                if s[j:i] in words and dfs(j):
                    return True
            return False
        return dfs(len(s))
```

```python
class Solution(object):
    def wordBreak(self, s, wordDict):
        max_len = max(map(len, wordDict))
        words = set(wordDict)
        
        n = len(s)
        f = [True] + [False] * n
        for i in range(1, n+1):
            for j in range(i-1, max(i-max_len-1, -1), -1):
                if f[j] and s[j:i] in words:
                    f[i] = True
                    break
        return f[n]
```

```python
class Solution:
    def countSubstrings(self, s: str) -> int:
        dp = [[False] * len(s) for _ in range(len(s))]
        result = 0
        for i in range(len(s) - 1, -1, -1):
            for j in range(i, len(s)):
                if s[i] == s[j]:
                    if j - i <= 1:
                        result += 1
                        dp[i][j] = True
                    elif dp[i + 1][j - 1]:
                        result += 1
                        dp[i][j] = True
        return result
```
