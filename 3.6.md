#### 排序

347 前K个高频元素

桶排序写法 也可以用堆排序 写堆然后循环更新

```python
class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        cnt_n = Counter(nums)
        max_cnt = max(cnt_n.values())

        buckets = [[] for _ in range(max_cnt+1)]

        for key, val in cnt_n.items():
            buckets[val].append(key)

        ans = []
        for bucket in reversed(buckets):
            ans+= bucket
            if len(ans) == k:
                return ans
```

#### 树形DP

打家劫舍III

```python
class Solution:
    def rob(self, root: Optional[TreeNode]) -> int:
        
        def dfs(node):
            if node is None:
                return 0, 0
            l_rob, l_not_rob = dfs(node.left)
            r_rob, r_not_rob = dfs(node.right)
            # 当前节点选 左右两个就不能选
            rob = l_not_rob + r_not_rob + node.val 
            # 当前节点不选 左右两个就可以选也可以不选取最大
            not_rob = max(l_not_rob, l_rob) + max(r_not_rob, r_rob)

            return rob, not_rob

        return max(dfs(root))
```
